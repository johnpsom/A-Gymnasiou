\lyseisE{Επανάληψη κεφαλαίου Α6}
\solr{Α6}{1} $\beta+7$, \quad $\beta-2$, \quad $5\beta$, \quad $\beta:4$.
\solr{Α6}{2} α) Ναι: $4+9=13$. β) Όχι: $3\cdot4=12\neq15$. γ) Ναι: $20-4=16$. δ) Όχι: $2\cdot4+1=9\neq10$.
\solr{Α6}{3} α) $21$ \quad β) $6{,}5$ \quad γ) $18$ \quad δ) $7$ \quad ε) $21$ \quad στ) $7$.
\solr{Α6}{4} α) $x=-7$ \quad β) $x=-10+3=-7$ \quad γ) $-x=-7$, $x=7$. Έλεγχοι: $-4\cdot(-7)=28$, $-7-3=-10$, $5-7=-2$.
\solr{Α6}{5} α) $5x=25$, $x=5$ \quad β) $4x=24$, $x=6$ \quad γ) $-3x=12$, $x=-4$ \quad δ) $0{,}5x=1{,}5$, $x=3$.
\solr{Α6}{6} α) $\frac{2}{3}x=4$, $x=6$ \quad β) $\frac{x}{4}=3$, $x=12$.
\solr{Α6}{7} $2x+10-3x+3=10$, $-x+13=10$, $-x=-3$, $x=3$. Έλεγχος: $2\cdot8-3\cdot2=10$.
\solr{Α6}{8} $6x-3-x-4=12$, $5x-7=12$, $5x=19$, $x=\frac{19}{5}=3{,}8$.
\solr{Α6}{9} Η πρώτη: $4x+8=4x+8$, ταυτότητα. Η δεύτερη: $3x+1=3x-4$, δηλαδή $1=-4$, αδύνατη.
\solr{Α6}{10} $3x+8=29$, $3x=21$, $x=7$.
\solr{Α6}{11} $3x+6=15$, $3x=9$, $x=3$ €.
\solr{Α6}{12} $x+(x+7)=33$, $2x=26$, $x=13$ €.
\solr{Α6}{13} Μήκος $x$: $2x+2\cdot8=54$, $2x=38$, $x=19$ cm.
\solr{Α6}{14} $2x-20=130$, $2x=150$, $x=75$ ετών.
\solr{Α6}{15} α) Σωστό. \quad β) Σωστό. \quad γ) Λάθος: είναι αδύνατη. \quad δ) Λάθος: δεν διαιρούμε με το $0$ (π.χ. $3\cdot0=5\cdot0$ αλλά $3\neq5$).
\solr{Α6}{16} α) Λάθος: η λύση είναι το $1$. \quad β) Σωστό. \quad γ) Σωστό. \quad δ) Σωστό: και οι δύο έχουν λύση το $4$.
